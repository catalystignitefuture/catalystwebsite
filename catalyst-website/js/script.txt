/* =========================================================
   2. CONTACT CONFIGURATION HELPERS
   ========================================================= */

function isPlaceholder(value) {

    const text = String(value || "").trim();

    return (
        !text ||
        /\[.*\]/.test(text) ||
        /YOUR/i.test(text) ||
        /X{2,}/i.test(text) ||
        /example\.com/i.test(text)
    );

}


function getWhatsAppDigits() {

    return String(CATALYST_CONFIG.whatsappNumber || "")
        .replace(/\D/g, "");

}


function isWhatsAppConfigured() {

    const rawNumber = String(
        CATALYST_CONFIG.whatsappNumber || ""
    ).trim();

    const digits = getWhatsAppDigits();

    return (
        !isPlaceholder(rawNumber) &&
        /^\d{8,15}$/.test(digits)
    );

}


function isPhoneConfigured() {

    const rawPhone = String(
        CATALYST_CONFIG.phoneDial || ""
    ).trim();

    const digits = rawPhone.replace(/\D/g, "");

    return (
        !isPlaceholder(rawPhone) &&
        /^\d{8,15}$/.test(digits)
    );

}


function isEmailConfigured() {

    const email = String(
        CATALYST_CONFIG.email || ""
    ).trim();

    return (
        !isPlaceholder(email) &&
        /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
    );

}


/* =========================================================
   3. REPLACE CONTACT PLACEHOLDERS ACROSS THE WEBSITE

   This replaces the original placeholder text in your pages.
   It preserves surrounding HTML, including contact icons.
   ========================================================= */

function replaceContactPlaceholders() {

    const replacements = [

        {
            token: "[YOUR PHONE / WHATSAPP]",
            value: CATALYST_CONFIG.phoneDisplay,
            href: isPhoneConfigured()
                ? "tel:" + CATALYST_CONFIG.phoneDial.replace(/[^\d+]/g, "")
                : null
        },

        {
            token: "[YOUR EMAIL]",
            value: CATALYST_CONFIG.email,
            href: isEmailConfigured()
                ? "mailto:" + CATALYST_CONFIG.email.trim()
                : null
        },

        {
            token: "[YOUR LOCATION]",
            value: CATALYST_CONFIG.location,
            href: null
        }

    ];


    const walker = document.createTreeWalker(
        document.body,
        NodeFilter.SHOW_TEXT
    );


    const textNodes = [];

    let currentNode;


    while ((currentNode = walker.nextNode())) {

        const parent = currentNode.parentElement;

        if (!parent) {
            continue;
        }

        if (
            parent.closest("script, style, textarea, input")
        ) {
            continue;
        }

        if (
            replacements.some(item =>
                currentNode.nodeValue.includes(item.token)
            )
        ) {

            textNodes.push(currentNode);

        }

    }


    textNodes.forEach(textNode => {

        const originalText = textNode.nodeValue;

        const fragment = document.createDocumentFragment();

        let remainingText = originalText;


        while (remainingText.length > 0) {

            let firstMatch = null;

            let firstIndex = -1;


            replacements.forEach(item => {

                const index = remainingText.indexOf(item.token);

                if (
                    index !== -1 &&
                    (firstIndex === -1 || index < firstIndex)
                ) {

                    firstIndex = index;
                    firstMatch = item;

                }

            });


            if (!firstMatch) {

                fragment.appendChild(
                    document.createTextNode(remainingText)
                );

                break;

            }


            if (firstIndex > 0) {

                fragment.appendChild(
                    document.createTextNode(
                        remainingText.slice(0, firstIndex)
                    )
                );

            }


            const canReplace =
                firstMatch.token === "[YOUR PHONE / WHATSAPP]"
                    ? !isPlaceholder(firstMatch.value)
                    : firstMatch.token === "[YOUR EMAIL]"
                        ? isEmailConfigured()
                        : !isPlaceholder(firstMatch.value);


            if (!canReplace) {

                fragment.appendChild(
                    document.createTextNode(firstMatch.token)
                );

            } else if (
                firstMatch.href &&
                !textNode.parentElement.closest("a")
            ) {

                const link = document.createElement("a");

                link.href = firstMatch.href;

                link.textContent = firstMatch.value;

                link.style.color = "inherit";
                link.style.textDecoration = "none";

                link.setAttribute(
                    "aria-label",
                    firstMatch.token === "[YOUR EMAIL]"
                        ? "Email CATALYST"
                        : "Call CATALYST"
                );

                fragment.appendChild(link);

            } else {

                fragment.appendChild(
                    document.createTextNode(firstMatch.value)
                );

            }


            remainingText =
                remainingText.slice(
                    firstIndex + firstMatch.token.length
                );

        }


        textNode.parentNode.replaceChild(
            fragment,
            textNode
        );

    });

}


/* =========================================================
   4. WHATSAPP BUTTONS

   Supports all links using:
   data-whatsapp-link
   ========================================================= */

function setupWhatsAppLinks() {

    const whatsappLinks = document.querySelectorAll(
        "[data-whatsapp-link], .contact-whatsapp"
    );


    whatsappLinks.forEach(link => {

        const message =
            link.getAttribute("data-whatsapp-message") ||
            "Hello CATALYST! I would like to discuss a project.";


        if (isWhatsAppConfigured()) {

            const whatsappURL =
                "https://wa.me/" +
                getWhatsAppDigits() +
                "?text=" +
                encodeURIComponent(message);


            link.href = whatsappURL;

            link.target = "_blank";

            link.rel = "noopener noreferrer";

            link.removeAttribute("aria-disabled");

            link.removeAttribute("title");

        } else {

            link.href = "#";

            link.removeAttribute("target");

            link.setAttribute("aria-disabled", "true");

            link.title =
                "Add your WhatsApp number in js/script.js first.";

        }


        /*
           Give a useful message if the number is still a
           placeholder, instead of opening a broken link.
        */

        link.addEventListener("click", function(event) {

            if (!isWhatsAppConfigured()) {

                event.preventDefault();

                alert(
                    "Please add your real WhatsApp number at the top of js/script.js before using this button."
                );

            }

        });

    });

}


/* =========================================================
   5. MOBILE NAVIGATION
   ========================================================= */

function setupMobileNavigation() {

    const menuButton = document.querySelector(
        ".mobile-menu-button"
    );

    const navLinks = document.querySelector(
        ".nav-links"
    );


    if (!menuButton || !navLinks) {
        return;
    }


    menuButton.setAttribute("aria-expanded", "false");


    menuButton.addEventListener("click", function() {

        const isOpen = navLinks.classList.toggle("active");

        menuButton.classList.toggle("active", isOpen);

        menuButton.setAttribute(
            "aria-expanded",
            String(isOpen)
        );

    });


    navLinks.querySelectorAll("a").forEach(link => {

        link.addEventListener("click", function() {

            navLinks.classList.remove("active");

            menuButton.classList.remove("active");

            menuButton.setAttribute(
                "aria-expanded",
                "false"
            );

        });

    });


    document.addEventListener("keydown", function(event) {

        if (event.key === "Escape") {

            navLinks.classList.remove("active");

            menuButton.classList.remove("active");

            menuButton.setAttribute(
                "aria-expanded",
                "false"
            );

        }

    });

}


/* =========================================================
   6. SMOOTH SCROLLING FOR LINKS ON THE CURRENT PAGE
   ========================================================= */

function setupSmoothScrolling() {

    document.querySelectorAll('a[href*="#"]').forEach(link => {

        link.addEventListener("click", function(event) {

            const href = link.getAttribute("href");

            if (!href || href === "#") {
                return;
            }


            let targetURL;

            try {

                targetURL = new URL(
                    href,
                    window.location.href
                );

            } catch (error) {

                return;

            }


            if (
                targetURL.origin !== window.location.origin ||
                targetURL.pathname !== window.location.pathname ||
                !targetURL.hash
            ) {

                return;

            }


            const target = document.querySelector(
                targetURL.hash
            );


            if (!target) {
                return;
            }


            event.preventDefault();


            target.scrollIntoView({
                behavior: "smooth",
                block: "start"
            });


            if (window.history && window.history.pushState) {

                window.history.pushState(
                    null,
                    "",
                    targetURL.hash
                );

            }

        });

    });

}


/* =========================================================
   7. SCROLL PROGRESS BAR
   ========================================================= */

function setupScrollProgress() {

    let progressBar = document.querySelector(
        ".catalyst-scroll-progress"
    );


    if (!progressBar) {

        progressBar = document.createElement("div");

        progressBar.className =
            "catalyst-scroll-progress";

        progressBar.setAttribute(
            "aria-hidden",
            "true"
        );

        document.body.appendChild(progressBar);

    }


    Object.assign(progressBar.style, {

        position: "fixed",

        top: "0",

        left: "0",

        width: "0%",

        height: "3px",

        zIndex: "99999",

        background:
            "linear-gradient(90deg, #00d4ff, #2f6bff)",

        boxShadow:
            "0 0 12px rgba(0, 212, 255, 0.45)",

        pointerEvents: "none",

        transition: "width 0.08s linear"

    });


    function updateProgress() {

        const scrollableHeight =
            document.documentElement.scrollHeight -
            window.innerHeight;


        if (scrollableHeight <= 0) {

            progressBar.style.width = "0%";

            return;

        }


        const percentage =
            (window.scrollY / scrollableHeight) * 100;


        progressBar.style.width =
            Math.min(100, Math.max(0, percentage)) + "%";

    }


    window.addEventListener(
        "scroll",
        updateProgress,
        { passive: true }
    );

    window.addEventListener(
        "resize",
        updateProgress
    );


    updateProgress();

}


/* =========================================================
   8. SCROLL REVEAL ANIMATIONS
   ========================================================= */

function setupScrollReveals() {

    const style = document.createElement("style");

    style.textContent = `

        .catalyst-reveal {
            opacity: 0;
            transform: translateY(22px);
            transition:
                opacity 0.65s ease,
                transform 0.65s ease;
        }

        .catalyst-reveal.catalyst-visible {
            opacity: 1;
            transform: translateY(0);
        }

        @media (prefers-reduced-motion: reduce) {
            .catalyst-reveal {
                opacity: 1;
                transform: none;
                transition: none;
            }

            html {
                scroll-behavior: auto !important;
            }
        }

    `;

    document.head.appendChild(style);


    const selectors = [

        ".service-card",

        ".work-card",

        ".gallery-item",

        ".portfolio-item",

        ".stat-card",

        ".why-card",

        ".feature-card",

        ".reason-card",

        ".video-info-card",

        ".video-type-card",

        ".video-process-card",

        ".small-site-feature",

        ".small-site-use-card",

        ".small-site-process-card",

        ".design-card",

        ".design-service-card",

        ".nfc-feature-card",

        ".website-feature-card"

    ];


    const elements = document.querySelectorAll(
        selectors.join(", ")
    );


    if (
        !("IntersectionObserver" in window) ||
        window.matchMedia(
            "(prefers-reduced-motion: reduce)"
        ).matches
    ) {

        elements.forEach(element => {

            element.classList.add(
                "catalyst-visible"
            );

        });

        return;

    }


    const observer = new IntersectionObserver(

        function(entries, activeObserver) {

            entries.forEach(entry => {

                if (entry.isIntersecting) {

                    entry.target.classList.add(
                        "catalyst-visible"
                    );

                    activeObserver.unobserve(
                        entry.target
                    );

                }

            });

        },

        {
            threshold: 0.12,
            rootMargin: "0px 0px -25px 0px"
        }

    );


    elements.forEach((element, index) => {

        element.classList.add("catalyst-reveal");

        element.style.transitionDelay =
            Math.min(index % 4, 3) * 70 + "ms";


        observer.observe(element);

    });

}


/* =========================================================
   9. ANIMATED HERO TYPING TEXT

   Uses data-typing-words in your HTML.
   Example: data-typing-words="GROW|STAND OUT|GET NOTICED"
   ========================================================= */

function setupTypingText() {

    const target = document.querySelector(
        "[data-typing-words]"
    );


    if (!target) {
        return;
    }


    if (
        window.matchMedia(
            "(prefers-reduced-motion: reduce)"
        ).matches
    ) {
        return;
    }


    const words = target
        .getAttribute("data-typing-words")
        .split("|")
        .map(word => word.trim())
        .filter(Boolean);


    if (words.length === 0) {
        return;
    }


    const originalText = target.textContent.trim();

    const punctuation = originalText.endsWith(".") ? "." : "";


    let wordIndex = 0;

    let characterIndex = 0;

    let deleting = false;


    function typeNextCharacter() {

        const currentWord = words[wordIndex];

        if (!currentWord) {
            return;
        }


        if (deleting) {

            characterIndex--;

        } else {

            characterIndex++;

        }


        target.textContent =
            currentWord.slice(0, characterIndex) +
            (
                characterIndex === currentWord.length
                    ? punctuation
                    : ""
            );


        let delay = deleting ? 45 : 90;


        if (!deleting && characterIndex === currentWord.length) {

            delay = 1350;

            deleting = true;

        } else if (deleting && characterIndex === 0) {

            deleting = false;

            wordIndex = (wordIndex + 1) % words.length;

            delay = 250;

        }


        window.setTimeout(
            typeNextCharacter,
            delay
        );

    }


    target.textContent = words[0] + punctuation;

    window.setTimeout(
        typeNextCharacter,
        1500
    );

}


/* =========================================================
   10. ANIMATED NUMBER COUNTERS

   Works with data-count or data-target attributes.
   Example: <span data-count="100" data-suffix="+">100+</span>
   ========================================================= */

function setupCounters() {

    const counters = document.querySelectorAll(
        "[data-count], [data-target]"
    );


    if (counters.length === 0) {
        return;
    }


    function animateCounter(element) {

        const rawTarget =
            element.getAttribute("data-count") ||
            element.getAttribute("data-target");


        const targetValue = Number(
            String(rawTarget).replace(/[^\d.-]/g, "")
        );


        if (!Number.isFinite(targetValue)) {
            return;
        }


        const suffix =
            element.getAttribute("data-suffix") || "";


        const prefix =
            element.getAttribute("data-prefix") || "";


        const duration = 1400;

        const startTime = performance.now();


        function frame(now) {

            const progress = Math.min(
                (now - startTime) / duration,
                1
            );


            const easedProgress =
                1 - Math.pow(1 - progress, 3);


            const currentValue = Math.round(
                targetValue * easedProgress
            );


            element.textContent =
                prefix +
                currentValue.toLocaleString() +
                suffix;


            if (progress < 1) {

                window.requestAnimationFrame(frame);

            }

        }


        window.requestAnimationFrame(frame);

    }


    if (!("IntersectionObserver" in window)) {

        counters.forEach(animateCounter);

        return;

    }


    const observer = new IntersectionObserver(

        function(entries, activeObserver) {

            entries.forEach(entry => {

                if (entry.isIntersecting) {

                    animateCounter(entry.target);

                    activeObserver.unobserve(
                        entry.target
                    );

                }

            });

        },

        {
            threshold: 0.4
        }

    );


    counters.forEach(counter => {

        observer.observe(counter);

    });

}


/* =========================================================
   11. GALLERY LIGHTBOX

   Click an image within #work or an image marked
   data-lightbox to see a larger version.
   ========================================================= */

function setupGalleryLightbox() {

    const galleryImages = document.querySelectorAll(
        '#work img, [data-lightbox]'
    );


    if (galleryImages.length === 0) {
        return;
    }


    const overlay = document.createElement("div");

    overlay.className = "catalyst-lightbox";


    Object.assign(overlay.style, {

        position: "fixed",

        inset: "0",

        display: "none",

        alignItems: "center",

        justifyContent: "center",

        padding: "24px",

        background: "rgba(2, 6, 18, 0.94)",

        backdropFilter: "blur(12px)",

        zIndex: "100000"

    });


    const image = document.createElement("img");

    image.alt = "";

    Object.assign(image.style, {

        maxWidth: "min(100%, 1100px)",

        maxHeight: "85vh",

        objectFit: "contain",

        borderRadius: "12px",

        boxShadow:
            "0 0 45px rgba(0, 212, 255, 0.14)"

    });


    const closeButton = document.createElement("button");

    closeButton.type = "button";

    closeButton.textContent = "×";

    closeButton.setAttribute(
        "aria-label",
        "Close image preview"
    );


    Object.assign(closeButton.style, {

        position: "absolute",

        top: "18px",

        right: "22px",

        width: "46px",

        height: "46px",

        fontSize: "32px",

        lineHeight: "1",

        color: "#ffffff",

        background: "rgba(255,255,255,0.08)",

        border: "1px solid rgba(0,212,255,0.25)",

        borderRadius: "50%",

        cursor: "pointer"

    });


    overlay.appendChild(image);

    overlay.appendChild(closeButton);

    document.body.appendChild(overlay);


    function openLightbox(sourceImage) {

        const source =
            sourceImage.currentSrc ||
            sourceImage.src;


        if (!source) {
            return;
        }


        image.src = source;

        image.alt =
            sourceImage.alt || "CATALYST portfolio image";


        overlay.style.display = "flex";

        document.body.style.overflow = "hidden";

        closeButton.focus();

    }


    function closeLightbox() {

        overlay.style.display = "none";

        image.removeAttribute("src");

        document.body.style.overflow = "";

    }


    galleryImages.forEach(galleryImage => {

        galleryImage.style.cursor = "zoom-in";

        galleryImage.addEventListener("click", function(event) {

            event.preventDefault();

            openLightbox(galleryImage);

        });

    });


    closeButton.addEventListener(
        "click",
        closeLightbox
    );


    overlay.addEventListener("click", function(event) {

        if (event.target === overlay) {

            closeLightbox();

        }

    });


    document.addEventListener("keydown", function(event) {

        if (
            event.key === "Escape" &&
            overlay.style.display === "flex"
        ) {

            closeLightbox();

        }

    });

}

    );

    }


    const style = document.createElement("style");

        "scroll",
        updateButton,
        { passive: true }
    );


    button.addEventListener("click", function() {

        window.scrollTo({
            top: 0,
            behavior: "smooth"
        });

    });


    updateButton();

}


/* =========================================================
   13. CONTACT FORM EMAIL HANDLING

   This uses the visitor's email app via a mailto link.
   It does NOT send data to a server.

   Works when your contact form has an identifiable form,
   name, email and message fields.
   ========================================================= */

function setupContactForm() {

    const form = document.querySelector(
        "#contactForm, .contact-form, form[data-contact-form]"
    );


    if (!form) {
        return;
    }


    form.addEventListener("submit", function(event) {

        event.preventDefault();


        if (!isEmailConfigured()) {

            alert(
                "Please add your real business email at the top of js/script.js before using the contact form."
            );

            return;

        }


        function readField(selectors) {

            for (const selector of selectors) {

                const element = form.querySelector(selector);

                if (element && "value" in element) {

                    return element.value.trim();

                }

            }

            return "";

        }


        const name = readField([

            '[name="name"]',
            '[name="fullName"]',
            "#name",
            'input[autocomplete="name"]'

        ]);


        const visitorEmail = readField([

            '[name="email"]',
            "#email",
            'input[type="email"]'

        ]);


        const phone = readField([

            '[name="phone"]',
            '[name="telephone"]',
            "#phone",
            'input[type="tel"]'

        ]);


        const service = readField([

            '[name="service"]',
            "#service",
            "select"

        ]);


        const message = readField([

            '[name="message"]',
            "#message",
            "textarea"

        ]);


        if (!message) {

            alert(
                "Please enter a message before submitting the form."
            );

            return;

        }


        const subject =
            "New Project Enquiry - CATALYST";


        const body = [

            "Hello CATALYST,",

            "",

            "I would like to discuss a project.",

            "",

            "Name: " + (name || "Not provided"),

            "Email: " + (visitorEmail || "Not provided"),

            "Phone: " + (phone || "Not provided"),

            "Service: " + (service || "Not specified"),

            "",

            "Message:",

            message

        ].join("\n");


        const mailtoURL =
            "mailto:" +
            CATALYST_CONFIG.email.trim() +
            "?subject=" +
            encodeURIComponent(subject) +
            "&body=" +
            encodeURIComponent(body);


        window.location.href = mailtoURL;

    });

}
